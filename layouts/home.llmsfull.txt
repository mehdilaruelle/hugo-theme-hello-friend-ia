{{- /*
  llms-full.txt: the pages llms.txt maps, in the same order, carrying their
  text instead of a link to it. One request instead of N. Each entry comes
  through the partial page.md.md uses, so the two cannot disagree.

  params.llmsFullLimit caps the entries, and when it bites the file says so: a
  truncated copy presenting itself as the whole site is the thing to avoid.
*/ -}}
{{- $desc := .Description | default site.Params.homeSubtitle | default site.Params.description -}}
{{- /* Posts first, then everything else — llms.txt order, not site-wide date
       order, which is why the notice below claims no recency. */ -}}
{{- $all := where site.RegularPages "Params.searchable" "!=" false -}}
{{- $posts := where $all "Section" "posts" -}}
{{- $other := $all | complement $posts -}}
{{- $ordered := slice -}}
{{- range $posts -}}{{- $ordered = $ordered | append . -}}{{- end -}}
{{- range $other -}}{{- $ordered = $ordered | append . -}}{{- end -}}
{{- $total := len $ordered -}}
{{- $limit := int (site.Params.llmsFullLimit | default 0) -}}
{{- if and (gt $limit 0) (lt $limit $total) -}}
  {{- $ordered = first $limit $ordered -}}
{{- end -}}
# {{ site.Title }}
{{ with $desc }}
> {{ . | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace }}
{{ end }}
{{- with .RenderShortcodes | strings.TrimSpace }}
{{ . }}
{{ end }}
{{- with site.Params.llmsNote }}
{{ . | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace }}
{{ end }}
{{- if lt (len $ordered) $total }}
{{- /* Kept, not Count: a Count key sends i18n looking for a plural form. */ -}}
{{ partial "i18n.html" (dict "key" "llmsFullTruncated" "args" (dict "Kept" (len $ordered) "Total" $total) "fallback" (printf "%d of %d pages, in the order llms.txt lists them." (len $ordered) $total)) }}
{{ end }}
{{- range $ordered }}
---

{{ partial "md/page.html" . }}
{{ end -}}

{{- /* Renders one content image, and is the single place that decides how. Both entry points call it: the image shortcode, and the cover on an article page. Plain markdown ![]() is left to Hugo. Resizing only happens when the file resolves as a resource — a page bundle, or assets/. A site whose images sit in static/ can mount that directory into assets to opt in; without it the image is still emitted, just unprocessed. Accepts: src, alt, position, style, page, and two options a caller above the fold needs: loading, to say eager where lazy would delay the largest paint, and maxWidth, to cap a picture that is never shown at the width articles get. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $page := .page -}} {{- $loading := .loading | default "lazy" -}} {{- $remote := partial "is-remote.html" $src -}} {{- $img := "" -}} {{- if not $remote -}} {{- with $page.Resources.GetMatch $src -}} {{- $img = . -}} {{- else -}} {{- with resources.Get (printf "_static%s" $src) -}}{{- $img = . -}}{{- else -}}{{- with resources.Get $src -}}{{- $img = . -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Animated GIFs lose their animation when resized, so leave them alone. An SVG reports ResourceType "image" and answers neither .Width nor .Resize, so asking either aborts the whole build. The two are separate questions: HEIC and HEIF carry their dimensions but cannot be resized, so measuring asks WithMeta and resizing asks Processable. A format that carries dimensions is still only answering for its kind. Where the file itself will not give them up Hugo returns 0, and width="0" collapses the picture, so the number is checked too. */ -}} {{- $maxWidth := .maxWidth | default site.Params.imageMaxWidth | default 1400 -}} {{- $rendered := $img -}} {{- $measurable := false -}} {{- if $img -}}{{- if reflect.IsImageResourceWithMeta $img -}} {{- $measurable = gt $img.Width 0 -}} {{- end -}}{{- end -}} {{- if and $img (reflect.IsImageResourceProcessable $img) -}} {{- if and (gt $img.Width $maxWidth) (ne $img.MediaType.SubType "gif") -}} {{- $rendered = $img.Resize (printf "%dx q85 webp" $maxWidth) -}} {{- end -}} {{- end -}} {{- $url := $src -}} {{- with $rendered -}} {{- $url = .RelPermalink -}} {{- else -}} {{- if not $remote -}}{{- $url = urls.JoinPath site.BaseURL $src -}}{{- end -}} {{- end -}} {{- if $src -}} {{- end -}}

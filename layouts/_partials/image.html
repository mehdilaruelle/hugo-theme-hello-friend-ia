{{- /* Renders one content image: the image shortcode and the article cover both call it. Plain markdown ![]() is left to Hugo. Resizing needs the file to resolve as a resource -- a page bundle, or assets/. A site keeping images in static/ can mount it into assets to opt in; without that the image is still emitted, just unprocessed. Takes src, alt, position, style, page, plus loading, fetchpriority, maxWidth, sizes and fill. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $page := .page -}} {{- $loading := .loading | default "lazy" -}} {{- /* No default: marking every picture high priority prioritises none. */ -}} {{- $fetchpriority := .fetchpriority -}} {{- /* The default is the measure of an article; anything narrower fills the viewport. A caller showing a fixed size passes its own. */ -}} {{- $sizes := .sizes | default site.Params.imageSizes | default "(max-width: 800px) 100vw, 800px" -}} {{- $remote := partial "is-remote.html" $src -}} {{- $img := "" -}} {{- if not $remote -}} {{- with $page.Resources.GetMatch $src -}} {{- $img = . -}} {{- else -}} {{- with resources.Get (printf "_static%s" $src) -}}{{- $img = . -}}{{- else -}}{{- with resources.Get $src -}}{{- $img = . -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Measuring and resizing are separate questions: an SVG answers neither .Width nor .Resize and asking aborts the build, while HEIC and HEIF carry their dimensions but cannot be resized. Hugo returns 0 where the file will not give its size up, and width="0" collapses the picture. */ -}} {{- $maxWidth := .maxWidth | default site.Params.imageMaxWidth | default 1400 -}} {{- $rendered := $img -}} {{- $measurable := false -}} {{- if $img -}}{{- if reflect.IsImageResourceWithMeta $img -}} {{- $measurable = gt $img.Width 0 -}} {{- end -}}{{- end -}} {{- /* srcset lets a phone pick a copy it can afford, and sizes tells it how wide the picture will be before any have loaded. Only widths smaller than the source are generated -- upscaling invents detail and pays for it. */ -}} {{- $widths := slice 480 800 1200 -}} {{- $srcset := slice -}} {{- $wantsFill := ne (.fill | default "") "" -}} {{- /* Resizing a GIF drops its animation, so one is left alone -- except in a box of a fixed shape, where the animation is a few dozen pixels wide and the untouched file is the whole download. */ -}} {{- $processable := and $img (reflect.IsImageResourceProcessable $img) (or $wantsFill (ne $img.MediaType.SubType "gif")) -}} {{- /* Capping the width fixes one axis only: a 900x600 cover capped at 156 is 156x104, and a square box stretches those 104 rows. Fill crops from the centre, which is what object-fit: cover was doing anyway. */ -}} {{- if and $processable $wantsFill -}} {{- $rendered = $img.Fill (printf "%s q85 webp" .fill) -}} {{- else if $processable -}} {{- $ceiling := cond (gt $img.Width $maxWidth) $maxWidth $img.Width -}} {{- $offered := slice -}} {{- range $widths -}} {{- if lt . $ceiling -}}{{- $offered = $offered | append . -}}{{- end -}} {{- end -}} {{- $offered = $offered | append $ceiling -}} {{- /* One width, and it is the source width: nothing to choose between and nothing to shrink, so the file is left alone rather than re-encoded. */ -}} {{- if and (eq (len $offered) 1) (eq $ceiling $img.Width) -}} {{- $srcset = slice -}} {{- else -}} {{- range $offered -}} {{- $copy := $img.Resize (printf "%dx q85 webp" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $copy.RelPermalink $copy.Width) -}} {{- if eq . $ceiling -}}{{- $rendered = $copy -}}{{- end -}} {{- end -}} {{- if eq (len $offered) 1 -}}{{- $srcset = slice -}}{{- end -}} {{- end -}} {{- end -}} {{- $url := $src -}} {{- with $rendered -}} {{- $url = .RelPermalink -}} {{- else -}} {{- if not $remote -}}{{- $url = urls.JoinPath site.BaseURL $src -}}{{- end -}} {{- end -}} {{- if $src -}} {{- end -}}

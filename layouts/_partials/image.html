{{- /* Renders one content image, and is the single place that decides how. Both entry points call it: the image shortcode, and the cover on an article page. Plain markdown ![]() is left to Hugo. Resizing only happens when the file resolves as a resource — a page bundle, or assets/. A site whose images sit in static/ can mount that directory into assets to opt in; without it the image is still emitted, just unprocessed. Accepts: src, alt, position, style, page, and three options: loading, to say eager where lazy would delay the largest paint; maxWidth, to cap a picture that is never shown at the width articles get; and sizes, to say how wide it will be shown when that is not the measure of an article. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $page := .page -}} {{- $loading := .loading | default "lazy" -}} {{- /* How wide the picture will be shown. The default is the measure of an article, and anything narrower than that fills the viewport. A caller showing a picture at a fixed size passes its own. */ -}} {{- $sizes := .sizes | default site.Params.imageSizes | default "(max-width: 800px) 100vw, 800px" -}} {{- $remote := partial "is-remote.html" $src -}} {{- $img := "" -}} {{- if not $remote -}} {{- with $page.Resources.GetMatch $src -}} {{- $img = . -}} {{- else -}} {{- with resources.Get (printf "_static%s" $src) -}}{{- $img = . -}}{{- else -}}{{- with resources.Get $src -}}{{- $img = . -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Animated GIFs lose their animation when resized, so leave them alone. An SVG reports ResourceType "image" and answers neither .Width nor .Resize, so asking either aborts the whole build. The two are separate questions: HEIC and HEIF carry their dimensions but cannot be resized, so measuring asks WithMeta and resizing asks Processable. A format that carries dimensions is still only answering for its kind. Where the file itself will not give them up Hugo returns 0, and width="0" collapses the picture, so the number is checked too. */ -}} {{- $maxWidth := .maxWidth | default site.Params.imageMaxWidth | default 1400 -}} {{- $rendered := $img -}} {{- $measurable := false -}} {{- if $img -}}{{- if reflect.IsImageResourceWithMeta $img -}} {{- $measurable = gt $img.Width 0 -}} {{- end -}}{{- end -}} {{- /* The widths to offer, and the copy the src points at. A phone asked to download a picture sized for a desktop pays for every pixel it will never show. srcset lets it pick, and sizes tells it how wide the picture will be before any of them have loaded. Only widths smaller than the source are generated: upscaling invents detail and costs bytes to do it. The largest is the cap, so the set is never wider than what the unresponsive src would have been. */ -}} {{- $widths := slice 480 800 1200 -}} {{- $srcset := slice -}} {{- $processable := and $img (reflect.IsImageResourceProcessable $img) (ne $img.MediaType.SubType "gif") -}} {{- if $processable -}} {{- $ceiling := cond (gt $img.Width $maxWidth) $maxWidth $img.Width -}} {{- $offered := slice -}} {{- range $widths -}} {{- if lt . $ceiling -}}{{- $offered = $offered | append . -}}{{- end -}} {{- end -}} {{- $offered = $offered | append $ceiling -}} {{- range $offered -}} {{- $copy := $img.Resize (printf "%dx q85 webp" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $copy.RelPermalink $copy.Width) -}} {{- if eq . $ceiling -}}{{- $rendered = $copy -}}{{- end -}} {{- end -}} {{- /* One width is the picture at its own size: nothing to choose between, so the attribute would only repeat what src already says. */ -}} {{- if eq (len $offered) 1 -}}{{- $srcset = slice -}}{{- end -}} {{- end -}} {{- $url := $src -}} {{- with $rendered -}} {{- $url = .RelPermalink -}} {{- else -}} {{- if not $remote -}}{{- $url = urls.JoinPath site.BaseURL $src -}}{{- end -}} {{- end -}} {{- if $src -}} {{- end -}}

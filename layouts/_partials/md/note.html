{{- /* The line after a link: the description, the summary when there is none, the empty string when there is neither. Flattened, because a newline ends a list item. htmlUnescape twice: once so truncate counts characters rather than entities, once to undo truncate's own escaping. */ -}} {{- $note := "" -}} {{- with .Description | default .Summary -}} {{- $note = . | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | truncate 200 "…" | htmlUnescape -}} {{- end -}} {{- return $note -}}

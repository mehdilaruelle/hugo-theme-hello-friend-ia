{{- /* Structured data as JSON-LD. Hugo's embedded schema.html already emits the same facts as microdata, but Google reads JSON-LD in preference to it. Home pages describe the site, dated single pages describe the article, and anything else is skipped rather than described badly. */ -}} {{- $author := partial "author.html" . -}} {{- $image := "" -}} {{- with .Params.images -}} {{- $image = index . 0 -}} {{- else -}} {{- with site.Params.images -}} {{- $image = index . 0 -}} {{- end -}} {{- end -}} {{- with $image -}} {{- $image = . | absURL -}} {{- end -}} {{- /* The Person behind the site, for the home page. A name on its own is a string; what makes it an entity a search engine can recognise is the evidence tying it to the same person elsewhere. sameAs carries the social profiles for that — the same claim rel="me" already makes on the links themselves. mailto: is an address, not a profile, so it is left out. jobTitle comes from params.author when it is written as a map. */ -}} {{- $jobTitle := "" -}} {{- with site.Params.author -}} {{- if reflect.IsMap . -}} {{- $jobTitle = .jobTitle -}} {{- end -}} {{- end -}} {{- $sameAs := slice -}} {{- range site.Params.social -}} {{- with .url -}} {{- if not (strings.HasPrefix (lower .) "mailto:") -}} {{- $sameAs = $sameAs | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- $person := dict "@type" "Person" -}} {{- with $author -}} {{- $person = merge $person (dict "@id" (printf "%s#person" site.Home.Permalink) "name" . "url" site.Home.Permalink) -}} {{- end -}} {{- with $jobTitle -}} {{- $person = merge $person (dict "jobTitle" .) -}} {{- end -}} {{- /* The portrait as a URL that exists. Resolving it as a resource is what publishes it: home.html renders a resized copy, and the original is emitted only if something asks for it. Taking the path as a string instead would name a file Hugo never wrote. A portrait outside assets/ is not a resource, and is published by Hugo as an ordinary static file, so the path stands. */ -}} {{- with site.Params.portrait.path -}} {{- $portrait := . | absURL -}} {{- with or (resources.Get (printf "_static%s" .)) (resources.Get .) -}} {{- $portrait = .Permalink -}} {{- end -}} {{- $person = merge $person (dict "image" $portrait) -}} {{- end -}} {{- with $sameAs -}} {{- $person = merge $person (dict "sameAs" .) -}} {{- end -}} {{- /* A page naming its own author is a different person from whoever owns the site, so it gets a name and nothing else: the site's profiles and job title are not theirs to claim. */ -}} {{- $authorNode := dict -}} {{- with $author -}} {{- $authorNode = cond (isset $.Params "author") (dict "@type" "Person" "name" .) $person -}} {{- end -}} {{- $data := dict -}} {{- if .IsHome -}} {{- $data = dict "@context" "https://schema.org" "@type" "WebSite" "url" .Permalink "name" site.Title "inLanguage" .Language.Lang -}} {{- with .Description | default site.Params.description -}} {{- $data = merge $data (dict "description" .) -}} {{- end -}} {{- with $author -}} {{- $data = merge $data (dict "author" $authorNode) -}} {{- end -}} {{- else if and .IsPage (not .Date.IsZero) -}} {{- $data = dict "@context" "https://schema.org" "@type" "BlogPosting" "@id" .Permalink "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "headline" (.Title | plainify | truncate 110 "") "datePublished" (.PublishDate.Format "2006-01-02T15:04:05Z07:00") "inLanguage" .Language.Lang "wordCount" .WordCount -}} {{- if not .Lastmod.IsZero -}} {{- $data = merge $data (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- /* Without a description of its own, fall back to a trimmed summary rather than the several hundred words .Summary can run to. */ -}} {{- with .Description | default (.Summary | plainify | chomp | truncate 300 "") -}} {{- $data = merge $data (dict "description" .) -}} {{- end -}} {{- with $author -}} {{- $data = merge $data (dict "author" $authorNode) -}} {{- end -}} {{- with $image -}} {{- $data = merge $data (dict "image" .) -}} {{- end -}} {{- with .Params.tags -}} {{- $data = merge $data (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- $data = merge $data (dict "publisher" (dict "@type" "Organization" "name" site.Title "url" site.Home.Permalink)) -}} {{- end -}} {{- /* safeJS is required: inside a script element html/template treats the value as JavaScript and would emit the object as a quoted string literal, which is not structured data any crawler would read. It is safe because jsonify escapes <, > and & to <, > and &, so no payload can close the element. */ -}} {{- with $data -}} {{- end -}}
